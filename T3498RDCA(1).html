Testing IDS Generalization Across Network Subnets¶

Research question¶

Does a machine-learning model detect attacks from traffic behavior, or does it rely on endpoint identity—such as IP addresses or subnets—that may not generalize to an unfamiliar network?

Notebook plan¶

  1. Explore the dataset and its labels (EDA).
  2. Preprocess features appropriately and explain each choice.
  3. Compare at least two models.
  4. Compare random cross-validation with subnet-grouped cross-validation.
  5. Interpret the results for real-world IDS deployment.

def get_source_network(ip):
    try:
        address = ipaddress.ip_address(ip)
        prefix = 24 if address.version == 4 else 64
        return str(ipaddress.ip_network(f"{address}/{prefix}", strict=False))
    except ValueError:
        return np.nan

def make_preprocessor(X):
    numeric_cols = X.select_dtypes(include=np.number).columns.tolist()
    categorical_cols = X.select_dtypes(exclude=np.number).columns.tolist()

    numeric_pipeline = Pipeline(steps=[
        ("imputer", SimpleImputer(strategy="median")),
        ("scaler", StandardScaler())
    ])

    categorical_pipeline = Pipeline(steps=[
        ("imputer", SimpleImputer(strategy="constant", fill_value="__MISSING__")),
        ("onehot", OneHotEncoder(
            handle_unknown="ignore",
            min_frequency=10
        ))
    ])

    preprocessor = ColumnTransformer(
        transformers=[
            ("numeric", numeric_pipeline, numeric_cols),
            ("categorical", categorical_pipeline, categorical_cols),
        ],
        remainder="drop"
    )

    return preprocessor, numeric_cols, categorical_cols


prep_identity, numeric_identity, categorical_identity = make_preprocessor(X_identity)

from sklearn.model_selection import StratifiedKFold, StratifiedGroupKFold

N_SPLITS = 5

# Group identical feature rows together for duplicate-aware CV.
# Targets and derived subnet columns are not included in X_identity.
duplicate_groups = pd.util.hash_pandas_object(
    X_identity, index=False
).astype("uint64")

cv_splits = {
    "Random stratified": list(
        StratifiedKFold(
            n_splits=N_SPLITS,
            shuffle=True,
            random_state=RANDOM_STATE
        ).split(X_identity, y)
    ),
    "Duplicate-group": list(
        StratifiedGroupKFold(
            n_splits=N_SPLITS,
            shuffle=True,
            random_state=RANDOM_STATE
        ).split(X_identity, y, groups=duplicate_groups)
    ),
    "Source-IP-group": list(
        StratifiedGroupKFold(
            n_splits=N_SPLITS,
            shuffle=True,
            random_state=RANDOM_STATE
        ).split(X_identity, y, groups=groups)
    ),
}

for cv_name, splits in cv_splits.items():
    print(f"\n{cv_name}")
    for fold, (train_idx, val_idx) in enumerate(splits, start=1):
        val_counts = y.iloc[val_idx].value_counts().sort_index().to_dict()

        if cv_name == "Random stratified":
            # Count validation rows whose identical feature row occurs in training.
            train_hashes = set(duplicate_groups.iloc[train_idx])
            overlap_rows = duplicate_groups.iloc[val_idx].isin(train_hashes).sum()
            extra = f", duplicate-overlap validation rows={overlap_rows:,}"
        elif cv_name == "Duplicate-group":
            extra = (
                ", duplicate groups shared across train/validation="
                f"{len(set(duplicate_groups.iloc[train_idx]) & set(duplicate_groups.iloc[val_idx]))}"
            )
        else:
            extra = (

    for feature_name, X_current in feature_sets.items():
        for model_name in ["Logistic Regression", "LightGBM"]:
            print(f"Running: {cv_name} | {feature_name} | {model_name}")

            for fold, (train_idx, val_idx) in enumerate(splits, start=1):
                X_train = X_current.iloc[train_idx]
                X_val = X_current.iloc[val_idx]
                y_train = y.iloc[train_idx]
                y_val = y.iloc[val_idx]

                preprocessor, _, _ = make_preprocessor(X_current)

                if model_name == "Logistic Regression":
                    model = LogisticRegression(
                        max_iter=1000,
                        class_weight="balanced",
                        solver="liblinear",
                        random_state=RANDOM_STATE,
                    )
                else:
                    model = LGBMClassifier(
                        n_estimators=250,
                        learning_rate=0.05,
                        num_leaves=31,
                        class_weight="balanced",
                        random_state=RANDOM_STATE,
                        n_jobs=-1,
                        verbosity=-1,
                    )

                pipeline = Pipeline(steps=[
                    ("prep", preprocessor),
                    ("model", model),
                ])

                pipeline.fit(X_train, y_train)

                # Probability of the malicious class (label 1)
                y_score = pipeline.predict_proba(X_val)[:, 1]
                y_pred = (y_score >= 0.5).astype(int)

                results.append({
                    "cv": cv_name,
                    "features": feature_name,
                    "model": model_name,
                    "fold": fold,
